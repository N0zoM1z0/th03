class SharedLock:
 def __init__(self,path):self.path=path;self.fd=None;self.expected=None
 @staticmethod
 def key(st):return (st.st_dev,st.st_ino,st.st_size,stat.S_IMODE(st.st_mode),st.st_nlink)
 def acquire(self):
  validate_parent(self.path);before=self.path.lstat()
  require(stat.S_ISREG(before.st_mode) and before.st_nlink==1 and stat.S_IMODE(before.st_mode)==0o600 and before.st_size==0,'unsafe/missing existing writer lock')
  fd=os.open(self.path,os.O_RDWR|os.O_NOFOLLOW)
  try:
   require(self.key(os.fstat(fd))==self.key(before),'writer lock changed during open')
   fcntl.flock(fd,fcntl.LOCK_EX|fcntl.LOCK_NB)
   self.fd=fd;self.expected=self.key(before);fd=None;self.check()
  finally:
   if fd is not None:os.close(fd)
 def check(self):
  require(self.fd is not None,'writer lock not held');validate_parent(self.path)
  current=self.path.lstat();held=os.fstat(self.fd)
  require(stat.S_ISREG(current.st_mode) and current.st_nlink==1 and self.key(current)==self.key(held)==self.expected,'held writer lock path/inode changed')
 def close(self):
  if self.fd is not None:
   try:fcntl.flock(self.fd,fcntl.LOCK_UN)
   finally:os.close(self.fd);self.fd=None

def logged_supervision(path,run,observed,open_log=None,retention=None):
 # Observe ownership before flush/close; retain result even when log retention fails.
 stream=None;outcome=None;errors=[];original=None
 try:
  stream=(open_log or (lambda p:p.open('xb')))(path)
  outcome=run(stream)
 except BaseException as error:original=error
 finally:
  if outcome is not None:observed(outcome)
  if stream is not None:
   try:
    with (retention('log flush') if retention else contextlib.nullcontext()):stream.flush();os.fsync(stream.fileno())
   except BaseException as error:errors.append('log flush/fsync: '+repr(error))
   try:
    with (retention('log close') if retention else contextlib.nullcontext()):stream.close()
   except BaseException as error:errors.append('log close: '+repr(error))
 return outcome,original,errors
