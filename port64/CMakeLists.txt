cmake_minimum_required(VERSION 3.16)
project(th03_port64 LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "TH03 port64 requires a 64-bit compiler")
endif()
# Shared BSD3 chip synthesis. Game resources and the rhythm ROM stay external.
add_library(th03-ymfm STATIC pmd/vendor/ymfm/src/ymfm_opn.cpp pmd/vendor/ymfm/src/ymfm_ssg.cpp pmd/vendor/ymfm/src/ymfm_adpcm.cpp)
target_compile_features(th03-ymfm PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(th03-ymfm PRIVATE -w)
endif()
add_library(th03-pmd STATIC pmd/pmd_sequence.cpp pmd/pmd_ssg_effects.cpp pmd/pmd_fm_effects.cpp pmd/pmd_musical_ssg.cpp pmd/pmd_rhythm.cpp pmd/pmd_musical_fm.cpp pmd/pmd_fm_player.cpp pmd/pmd_timer_player.cpp pmd/pmd_clock.cpp pmd/pmd_pcm.cpp)
target_compile_features(th03-pmd PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(th03-pmd PRIVATE $<$<CONFIG:Debug>:-O2>)
    target_compile_options(th03-ymfm PRIVATE $<$<CONFIG:Debug>:-O2>)
    set_source_files_properties(pmd/pmd_pcm.cpp PROPERTIES COMPILE_OPTIONS -Wno-unused-parameter)
endif()
target_include_directories(th03-pmd PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(th03-pmd PUBLIC th03-ymfm)
add_library(th03-portable STATIC assets.cpp pi_image.cpp menu.cpp graphics.cpp raster.cpp motion.cpp resident.cpp selection.cpp font.cpp stage_intro.cpp win.cpp score.cpp score_menu.cpp cutscene.cpp staff.cpp music.cpp configuration.cpp sound.cpp round.cpp enemy.cpp hud_intro.cpp collision.cpp bullet.cpp combat.cpp player.cpp cpu.cpp charge.cpp bomb.cpp background.cpp extra.cpp gauge_attacks.cpp boss.cpp warning.cpp battle_graphics.cpp results.cpp round_loop.cpp application.cpp)
target_compile_features(th03-portable PUBLIC cxx_std_17)
target_include_directories(th03-portable PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(th03-portable PUBLIC th03-pmd)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(th03-portable PUBLIC stdc++fs)
endif()
add_executable(th03-port64 main.cpp view.cpp audio_output.cpp audio_device.cpp)
target_link_libraries(th03-port64 PRIVATE th03-portable)
if(WIN32)
    target_link_libraries(th03-port64 PRIVATE user32 gdi32 winmm)
    if(MINGW)
        target_link_options(th03-port64 PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    target_link_libraries(th03-port64 PRIVATE PkgConfig::SDL2)
endif()
enable_testing()
add_executable(th03-port64-checks checks.cpp)
target_link_libraries(th03-port64-checks PRIVATE th03-portable)
if(MINGW)
    target_link_options(th03-port64-checks PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME portable-contracts COMMAND th03-port64-checks)

add_executable(th03-pmd-pcm-checks check_pcm.cpp)
target_link_libraries(th03-pmd-pcm-checks PRIVATE th03-pmd)
if(MINGW)
    target_link_options(th03-pmd-pcm-checks PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME pmd-pcm-contracts COMMAND th03-pmd-pcm-checks)

add_executable(th03-audio-device-checks check_audio_device.cpp audio_output.cpp audio_device.cpp)
target_link_libraries(th03-audio-device-checks PRIVATE th03-pmd)
if(WIN32)
    target_link_libraries(th03-audio-device-checks PRIVATE winmm)
    if(MINGW)
        target_link_options(th03-audio-device-checks PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
else()
    target_link_libraries(th03-audio-device-checks PRIVATE PkgConfig::SDL2)
endif()
add_test(NAME audio-device-contracts COMMAND th03-audio-device-checks)
